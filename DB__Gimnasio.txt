-- BASE DE DATOS: GIMNASIO CONTROL TOTAL

-- =============================
-- TABLA: Administrativos
-- =============================
CREATE TABLE administrativos (
    id_admin SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    usuario VARCHAR(50) UNIQUE NOT NULL,
    contraseña VARCHAR(255) NOT NULL, -- se recomienda encriptar
    fecha_contratacion DATE NOT NULL DEFAULT CURRENT_DATE
);

-- =============================
-- TABLA: Usuarios (Clientes)
-- =============================
CREATE TABLE usuarios (
    id_usuario SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE,
    telefono VARCHAR(20),
    email VARCHAR(100) UNIQUE,
    fecha_registro DATE NOT NULL DEFAULT CURRENT_DATE,
    registrado_por INT NOT NULL,
    CONSTRAINT fk_registrado_por FOREIGN KEY (registrado_por) REFERENCES administrativos(id_admin)
);

-- =============================
-- TABLA: Membresías
-- =============================
CREATE TABLE membresias (
    id_membresia SERIAL PRIMARY KEY,
    tipo VARCHAR(50) NOT NULL,
    duracion_dias INT NOT NULL,
    precio DECIMAL(10,2) NOT NULL
);

-- =============================
-- TABLA: Registro de Membresías
-- =============================
CREATE TABLE registro_membresias (
    id_registro SERIAL PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_membresia INT NOT NULL,
    id_admin INT NOT NULL,
    fecha_inicio DATE NOT NULL DEFAULT CURRENT_DATE,
    fecha_fin DATE,
    activo BOOLEAN DEFAULT TRUE,
    CONSTRAINT fk_reg_usuario FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario),
    CONSTRAINT fk_reg_membresia FOREIGN KEY (id_membresia) REFERENCES membresias(id_membresia),
    CONSTRAINT fk_reg_admin FOREIGN KEY (id_admin) REFERENCES administrativos(id_admin)
);

-- Trigger para calcular fecha_fin automáticamente
CREATE OR REPLACE FUNCTION calcular_fecha_fin()
RETURNS TRIGGER AS $$
BEGIN
    SELECT NEW.fecha_inicio + (m.duracion_dias || ' days')::interval
    INTO NEW.fecha_fin
    FROM membresias m
    WHERE m.id_membresia = NEW.id_membresia;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_calcular_fecha_fin
BEFORE INSERT ON registro_membresias
FOR EACH ROW
EXECUTE FUNCTION calcular_fecha_fin();

-- =============================
-- TABLA: Cajas
-- =============================
CREATE TABLE cajas (
    id_caja SERIAL PRIMARY KEY,
    descripcion VARCHAR(100),
    fecha_apertura DATE NOT NULL DEFAULT CURRENT_DATE,
    saldo_inicial DECIMAL(10,2) DEFAULT 0,
    saldo_actual DECIMAL(10,2) DEFAULT 0,
    abierta BOOLEAN DEFAULT TRUE
);

-- =============================
-- TABLA: Pagos (modificada con id_caja)
-- =============================
CREATE TABLE pagos (
    id_pago SERIAL PRIMARY KEY,
    id_registro INT NOT NULL,
    id_admin INT NOT NULL,
    id_caja INT NOT NULL DEFAULT 1 REFERENCES cajas(id_caja),
    fecha_pago DATE NOT NULL DEFAULT CURRENT_DATE,
    monto_pagado DECIMAL(10,2) NOT NULL,
    estado_pago VARCHAR(20) CHECK (estado_pago IN ('Completo', 'Parcial', 'Pendiente')),
    CONSTRAINT fk_pago_registro FOREIGN KEY (id_registro) REFERENCES registro_membresias(id_registro),
    CONSTRAINT fk_pago_admin FOREIGN KEY (id_admin) REFERENCES administrativos(id_admin)
);

-- =============================
-- TABLA: Productos
-- =============================
CREATE TABLE productos (
    id_producto SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    precio DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL DEFAULT 0
);

-- =============================
-- TABLA: Ventas
-- =============================
CREATE TABLE ventas (
    id_venta SERIAL PRIMARY KEY,
    id_usuario INT NOT NULL REFERENCES usuarios(id_usuario),
    id_admin INT NOT NULL REFERENCES administrativos(id_admin),
    id_caja INT NOT NULL REFERENCES cajas(id_caja),
    fecha_venta DATE DEFAULT CURRENT_DATE,
    total DECIMAL(10,2) NOT NULL
);

-- =============================
-- TABLA: Detalle Ventas
-- =============================
CREATE TABLE detalle_ventas (
    id_detalle SERIAL PRIMARY KEY,
    id_venta INT NOT NULL REFERENCES ventas(id_venta),
    id_producto INT NOT NULL REFERENCES productos(id_producto),
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) GENERATED ALWAYS AS (cantidad * precio_unitario) STORED
);

-- =============================
-- TABLA: Movimientos de Caja
-- =============================
CREATE TABLE movimientos_caja (
    id_movimiento SERIAL PRIMARY KEY,
    id_caja INT NOT NULL REFERENCES cajas(id_caja),
    id_admin INT NOT NULL REFERENCES administrativos(id_admin),
    tipo_movimiento VARCHAR(20) CHECK (tipo_movimiento IN ('Ingreso', 'Egreso')) NOT NULL,
    descripcion TEXT,
    monto DECIMAL(10,2) NOT NULL,
    fecha_movimiento DATE DEFAULT CURRENT_DATE,
    origen VARCHAR(20) CHECK (origen IN ('Pago', 'Venta', 'Desembolso', 'Reembolso', 'Otro')),
    id_referencia INT
);

-- ==========================================================
-- TRIGGER: reflejar pagos en cajas
-- ==========================================================
CREATE OR REPLACE FUNCTION reflejar_pago_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Obtener nombre completo del cliente
    SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
    FROM usuarios u
    JOIN registro_membresias r ON r.id_usuario = u.id_usuario
    WHERE r.id_registro = NEW.id_registro;

    -- Actualizar el saldo de la caja
    UPDATE cajas
    SET saldo_actual = saldo_actual + NEW.monto_pagado
    WHERE id_caja = NEW.id_caja;

    -- Registrar el movimiento en la tabla de movimientos
    INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
    VALUES (NEW.id_caja, NEW.id_admin, 'Ingreso', 'Pago de membresía de ' || nombre_cliente, NEW.monto_pagado, 'Pago', NEW.id_pago);

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_pago_en_caja
AFTER INSERT ON pagos
FOR EACH ROW
EXECUTE FUNCTION reflejar_pago_en_caja();

-- ==========================================================
-- TRIGGER: reflejar ventas en cajas
-- ==========================================================
CREATE OR REPLACE FUNCTION reflejar_venta_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Obtener nombre completo del cliente
    SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
    FROM usuarios u
    WHERE u.id_usuario = NEW.id_usuario;

    -- Actualizar el saldo de la caja
    UPDATE cajas
    SET saldo_actual = saldo_actual + NEW.total
    WHERE id_caja = NEW.id_caja;

    -- Registrar el movimiento en la tabla de movimientos
    INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
    VALUES (NEW.id_caja, NEW.id_admin, 'Ingreso', 'Venta de productos a ' || nombre_cliente, NEW.total, 'Venta', NEW.id_venta);

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_venta_en_caja
AFTER INSERT ON ventas
FOR EACH ROW
EXECUTE FUNCTION reflejar_venta_en_caja();

CREATE OR REPLACE FUNCTION calcular_fecha_fin()
RETURNS TRIGGER AS $$
BEGIN
    -- Calcular fecha_fin basado en fecha_inicio y duración de la membresía
    -- FIX: Cambiar ::interval::date por ::interval solamente
    SELECT NEW.fecha_inicio + (m.duracion_dias || ' days')::interval
    INTO NEW.fecha_fin
    FROM membresias m
    WHERE m.id_membresia = NEW.id_membresia;
    
    -- Si no se encuentra la membresía, usar default de 30 días
    IF NEW.fecha_fin IS NULL THEN
        NEW.fecha_fin := NEW.fecha_inicio + interval '30 days';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger para INSERT
CREATE TRIGGER tg_calcular_fecha_fin_insert
BEFORE INSERT ON registro_membresias
FOR EACH ROW
EXECUTE FUNCTION calcular_fecha_fin();

-- Trigger para UPDATE (cuando cambie fecha_inicio o id_membresia)
CREATE TRIGGER tg_calcular_fecha_fin_update
BEFORE UPDATE ON registro_membresias
FOR EACH ROW
WHEN (OLD.fecha_inicio IS DISTINCT FROM NEW.fecha_inicio OR OLD.id_membresia IS DISTINCT FROM NEW.id_membresia)
EXECUTE FUNCTION calcular_fecha_fin();

-- ==========================================================

-- Trigger para restar de caja cuando se elimina (soft delete) un pago
CREATE OR REPLACE FUNCTION reflejar_eliminacion_pago_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Solo procesar si el pago se está desactivando (soft delete)
    IF OLD.activo = TRUE AND NEW.activo = FALSE THEN
        -- Obtener nombre completo del cliente
        SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
        FROM usuarios u
        JOIN registro_membresias r ON r.id_usuario = u.id_usuario
        WHERE r.id_registro = OLD.id_registro;

        -- Restar el monto de la caja
        UPDATE cajas
        SET saldo_actual = saldo_actual - OLD.monto_pagado
        WHERE id_caja = OLD.id_caja;

        -- Registrar el movimiento de egreso por eliminación
        INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
        VALUES (OLD.id_caja, NEW.id_admin, 'Egreso', 'Eliminación de pago de membresía de ' || nombre_cliente, OLD.monto_pagado, 'Reembolso', OLD.id_pago);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_eliminacion_pago_en_caja
AFTER UPDATE ON pagos
FOR EACH ROW
EXECUTE FUNCTION reflejar_eliminacion_pago_en_caja();
