-- BASE DE DATOS: GIMNASIO CONTROL TOTAL

-- =============================
-- TABLA: Administrativos
-- =============================
CREATE TABLE administrativos (
    id_admin SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    usuario VARCHAR(50) UNIQUE NOT NULL,
    contraseña VARCHAR(255) NOT NULL, -- se recomienda encriptar
    fecha_contratacion DATE NOT NULL DEFAULT CURRENT_DATE
);

-- =============================
-- TABLA: Usuarios (Clientes)
-- =============================
CREATE TABLE usuarios (
    id_usuario SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE,
    telefono VARCHAR(20),
    email VARCHAR(100) UNIQUE,
    fecha_registro DATE NOT NULL DEFAULT CURRENT_DATE,
    registrado_por INT NOT NULL,
    CONSTRAINT fk_registrado_por FOREIGN KEY (registrado_por) REFERENCES administrativos(id_admin)
);

-- =============================
-- TABLA: Membresías
-- =============================
CREATE TABLE membresias (
    id_membresia SERIAL PRIMARY KEY,
    tipo VARCHAR(50) NOT NULL,
    duracion_dias INT NOT NULL,
    precio DECIMAL(10,2) NOT NULL
);

-- =============================
-- TABLA: Registro de Membresías
-- =============================
CREATE TABLE registro_membresias (
    id_registro SERIAL PRIMARY KEY,
    id_usuario INT NOT NULL,
    id_membresia INT NOT NULL,
    id_admin INT NOT NULL,
    fecha_inicio DATE NOT NULL DEFAULT CURRENT_DATE,
    fecha_fin DATE,
    activo BOOLEAN DEFAULT TRUE,
    CONSTRAINT fk_reg_usuario FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario),
    CONSTRAINT fk_reg_membresia FOREIGN KEY (id_membresia) REFERENCES membresias(id_membresia),
    CONSTRAINT fk_reg_admin FOREIGN KEY (id_admin) REFERENCES administrativos(id_admin)
);

-- Trigger para calcular fecha_fin automáticamente
CREATE OR REPLACE FUNCTION calcular_fecha_fin()
RETURNS TRIGGER AS $$
BEGIN
    SELECT NEW.fecha_inicio + (m.duracion_dias || ' days')::interval
    INTO NEW.fecha_fin
    FROM membresias m
    WHERE m.id_membresia = NEW.id_membresia;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_calcular_fecha_fin
BEFORE INSERT ON registro_membresias
FOR EACH ROW
EXECUTE FUNCTION calcular_fecha_fin();

-- =============================
-- TABLA: Cajas
-- =============================
CREATE TABLE cajas (
    id_caja SERIAL PRIMARY KEY,
    descripcion VARCHAR(100),
    fecha_apertura DATE NOT NULL DEFAULT CURRENT_DATE,
    saldo_inicial DECIMAL(10,2) DEFAULT 0,
    saldo_actual DECIMAL(10,2) DEFAULT 0,
    abierta BOOLEAN DEFAULT TRUE
);

-- =============================
-- TABLA: Pagos (modificada con id_caja)
-- =============================
CREATE TABLE pagos (
    id_pago SERIAL PRIMARY KEY,
    id_registro INT NOT NULL,
    id_admin INT NOT NULL,
    id_caja INT NOT NULL DEFAULT 1 REFERENCES cajas(id_caja),
    fecha_pago DATE NOT NULL DEFAULT CURRENT_DATE,
    monto_pagado DECIMAL(10,2) NOT NULL,
    estado_pago VARCHAR(20) CHECK (estado_pago IN ('Completo', 'Parcial', 'Pendiente')),
    CONSTRAINT fk_pago_registro FOREIGN KEY (id_registro) REFERENCES registro_membresias(id_registro),
    CONSTRAINT fk_pago_admin FOREIGN KEY (id_admin) REFERENCES administrativos(id_admin)
);

-- =============================
-- TABLA: Productos
-- =============================
CREATE TABLE productos (
    id_producto SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    precio DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL DEFAULT 0
);

-- =============================
-- TABLA: Ventas
-- =============================
CREATE TABLE ventas (
    id_venta SERIAL PRIMARY KEY,
    id_usuario INT NOT NULL REFERENCES usuarios(id_usuario),
    id_admin INT NOT NULL REFERENCES administrativos(id_admin),
    id_caja INT NOT NULL REFERENCES cajas(id_caja),
    fecha_venta DATE DEFAULT CURRENT_DATE,
    total DECIMAL(10,2) NOT NULL
);

-- =============================
-- TABLA: Detalle Ventas
-- =============================
CREATE TABLE detalle_ventas (
    id_detalle SERIAL PRIMARY KEY,
    id_venta INT NOT NULL REFERENCES ventas(id_venta),
    id_producto INT NOT NULL REFERENCES productos(id_producto),
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) GENERATED ALWAYS AS (cantidad * precio_unitario) STORED
);

-- =============================
-- TABLA: Movimientos de Caja
-- =============================
CREATE TABLE movimientos_caja (
    id_movimiento SERIAL PRIMARY KEY,
    id_caja INT NOT NULL REFERENCES cajas(id_caja),
    id_admin INT NOT NULL REFERENCES administrativos(id_admin),
    tipo_movimiento VARCHAR(20) CHECK (tipo_movimiento IN ('Ingreso', 'Egreso')) NOT NULL,
    descripcion TEXT,
    monto DECIMAL(10,2) NOT NULL,
    fecha_movimiento DATE DEFAULT CURRENT_DATE,
    origen VARCHAR(20) CHECK (origen IN ('Pago', 'Venta', 'Desembolso', 'Reembolso', 'Otro')),
    id_referencia INT
);

-- ==========================================================
-- TRIGGER: reflejar pagos en cajas
-- ==========================================================
CREATE OR REPLACE FUNCTION reflejar_pago_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Obtener nombre completo del cliente
    SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
    FROM usuarios u
    JOIN registro_membresias r ON r.id_usuario = u.id_usuario
    WHERE r.id_registro = NEW.id_registro;

    -- Actualizar el saldo de la caja
    UPDATE cajas
    SET saldo_actual = saldo_actual + NEW.monto_pagado
    WHERE id_caja = NEW.id_caja;

    -- Registrar el movimiento en la tabla de movimientos
    INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
    VALUES (NEW.id_caja, NEW.id_admin, 'Ingreso', 'Pago de membresía de ' || nombre_cliente, NEW.monto_pagado, 'Pago', NEW.id_pago);

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_pago_en_caja
AFTER INSERT ON pagos
FOR EACH ROW
EXECUTE FUNCTION reflejar_pago_en_caja();

-- ==========================================================
-- TRIGGER: reflejar ventas en cajas
-- ==========================================================
CREATE OR REPLACE FUNCTION reflejar_venta_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Obtener nombre completo del cliente
    SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
    FROM usuarios u
    WHERE u.id_usuario = NEW.id_usuario;

    -- Actualizar el saldo de la caja
    UPDATE cajas
    SET saldo_actual = saldo_actual + NEW.total
    WHERE id_caja = NEW.id_caja;

    -- Registrar el movimiento en la tabla de movimientos
    INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
    VALUES (NEW.id_caja, NEW.id_admin, 'Ingreso', 'Venta de productos a ' || nombre_cliente, NEW.total, 'Venta', NEW.id_venta);

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_venta_en_caja
AFTER INSERT ON ventas
FOR EACH ROW
EXECUTE FUNCTION reflejar_venta_en_caja();

CREATE OR REPLACE FUNCTION calcular_fecha_fin()
RETURNS TRIGGER AS $$
BEGIN
    -- Calcular fecha_fin basado en fecha_inicio y duración de la membresía
    -- FIX: Cambiar ::interval::date por ::interval solamente
    SELECT NEW.fecha_inicio + (m.duracion_dias || ' days')::interval
    INTO NEW.fecha_fin
    FROM membresias m
    WHERE m.id_membresia = NEW.id_membresia;
    
    -- Si no se encuentra la membresía, usar default de 30 días
    IF NEW.fecha_fin IS NULL THEN
        NEW.fecha_fin := NEW.fecha_inicio + interval '30 days';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger para INSERT
CREATE TRIGGER tg_calcular_fecha_fin_insert
BEFORE INSERT ON registro_membresias
FOR EACH ROW
EXECUTE FUNCTION calcular_fecha_fin();

-- Trigger para UPDATE (cuando cambie fecha_inicio o id_membresia)
CREATE TRIGGER tg_calcular_fecha_fin_update
BEFORE UPDATE ON registro_membresias
FOR EACH ROW
WHEN (OLD.fecha_inicio IS DISTINCT FROM NEW.fecha_inicio OR OLD.id_membresia IS DISTINCT FROM NEW.id_membresia)
EXECUTE FUNCTION calcular_fecha_fin();

-- ==========================================================
-- TRIGGERS PARA REFLEJAR ELIMINACIONES EN CAJAS
--
-- ADVERTENCIA (2026-09): estos dos triggers NO están instalados en la
-- base de datos real (verificado contra information_schema.triggers).
-- La eliminación de pagos/ventas ya la maneja el código de la app
-- (src/app/api/pagos/[id]/route.js y .../ventas/[id]/route.js: restan
-- de la caja, crean el movimiento de egreso y marcan el registro como
-- inactivo, todo dentro de una transacción con bloqueo de fila).
-- NO ejecutar este bloque contra la base real: si se instalan estos
-- triggers, cada eliminación de pago/venta pasaría a descontarse DOS
-- VECES de la caja (una por el código de la app, otra por el trigger),
-- dejando el saldo mal sin ningún error visible.
-- ==========================================================

-- Trigger para restar de caja cuando se elimina (soft delete) un pago
CREATE OR REPLACE FUNCTION reflejar_eliminacion_pago_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Solo procesar si el pago se está desactivando (soft delete)
    IF OLD.activo = TRUE AND NEW.activo = FALSE THEN
        -- Obtener nombre completo del cliente
        SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
        FROM usuarios u
        JOIN registro_membresias r ON r.id_usuario = u.id_usuario
        WHERE r.id_registro = OLD.id_registro;

        -- Restar el monto de la caja
        UPDATE cajas
        SET saldo_actual = saldo_actual - OLD.monto_pagado
        WHERE id_caja = OLD.id_caja;

        -- Registrar el movimiento de egreso por eliminación
        INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
        VALUES (OLD.id_caja, NEW.id_admin, 'Egreso', 'Eliminación de pago de membresía de ' || nombre_cliente, OLD.monto_pagado, 'Reembolso', OLD.id_pago);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_eliminacion_pago_en_caja
AFTER UPDATE ON pagos
FOR EACH ROW
EXECUTE FUNCTION reflejar_eliminacion_pago_en_caja();

-- Trigger para restar de caja cuando se elimina (soft delete) una venta
CREATE OR REPLACE FUNCTION reflejar_eliminacion_venta_en_caja()
RETURNS TRIGGER AS $$
DECLARE
    nombre_cliente TEXT;
BEGIN
    -- Solo procesar si la venta se está desactivando (soft delete)
    IF OLD.activo = TRUE AND NEW.activo = FALSE THEN
        -- Obtener nombre completo del cliente
        SELECT u.nombre || ' ' || u.apellido INTO nombre_cliente
        FROM usuarios u
        WHERE u.id_usuario = OLD.id_usuario;

        -- Restar el total de la caja
        UPDATE cajas
        SET saldo_actual = saldo_actual - OLD.total
        WHERE id_caja = OLD.id_caja;

        -- Registrar el movimiento de egreso por eliminación
        INSERT INTO movimientos_caja (id_caja, id_admin, tipo_movimiento, descripcion, monto, origen, id_referencia)
        VALUES (OLD.id_caja, NEW.id_admin, 'Egreso', 'Eliminación de venta de productos a ' || nombre_cliente, OLD.total, 'Reembolso', OLD.id_venta);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_reflejar_eliminacion_venta_en_caja
AFTER UPDATE ON ventas
FOR EACH ROW
EXECUTE FUNCTION reflejar_eliminacion_venta_en_caja();

-- APERTURA / CIERRE FORMAL DE CAJAS CON ARQUEO (2026-09)
-- ==========================================================
-- Antes de esto, "cajas" solo tenía saldo_inicial/saldo_actual/abierta: abrir
-- o cerrar una caja era simplemente togglear el booleano `abierta`, sin
-- registrar quién lo hizo, cuándo se cerró, ni comparar el saldo calculado
-- por el sistema contra el dinero contado físicamente (arqueo).
--
-- IMPORTANTE: el sistema sigue permitiendo VARIAS cajas abiertas al mismo
-- tiempo (ej. "Caja Efectivo" y "Caja Qr" abiertas en paralelo), porque el
-- pago mixto reparte un cobro entre dos cajas distintas y ambas deben poder
-- estar abiertas a la vez. La única regla nueva es: no se puede reabrir una
-- caja con la MISMA descripción mientras ya exista una abierta con ese
-- mismo nombre (ver índice único parcial más abajo). No es "una sola caja
-- abierta en todo el sistema".
--
-- No se elimina la columna `abierta` (se mantiene sincronizada con `estado`
-- por el código de la app) para no romper nada que todavía la lea.

ALTER TABLE cajas
ADD COLUMN IF NOT EXISTS fecha_cierre TIMESTAMP,
ADD COLUMN IF NOT EXISTS saldo_esperado DECIMAL(10,2),
ADD COLUMN IF NOT EXISTS saldo_contado DECIMAL(10,2),
ADD COLUMN IF NOT EXISTS diferencia DECIMAL(10,2),
ADD COLUMN IF NOT EXISTS id_admin_apertura INT REFERENCES administrativos(id_admin),
ADD COLUMN IF NOT EXISTS id_admin_cierre INT REFERENCES administrativos(id_admin),
ADD COLUMN IF NOT EXISTS estado VARCHAR(20) NOT NULL DEFAULT 'ABIERTA';

-- Constraints agregados por separado (no todos los motores/versiones
-- soportan CHECK inline en ADD COLUMN de forma portable, y así se puede
-- volver a correr el bloque sin duplicar el constraint).
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'chk_cajas_estado'
  ) THEN
    ALTER TABLE cajas ADD CONSTRAINT chk_cajas_estado CHECK (estado IN ('ABIERTA', 'CERRADA'));
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'chk_cajas_saldo_contado_no_negativo'
  ) THEN
    ALTER TABLE cajas ADD CONSTRAINT chk_cajas_saldo_contado_no_negativo CHECK (saldo_contado IS NULL OR saldo_contado >= 0);
  END IF;

  -- Esta ya existe en la base real (ver src/lib/db/erroresCaja.js, que la
  -- traduce a un mensaje legible) pero nunca quedó documentada en este
  -- archivo -- se deja constancia acá por si se recrea la base desde cero.
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'chk_cajas_saldo_no_negativo'
  ) THEN
    ALTER TABLE cajas ADD CONSTRAINT chk_cajas_saldo_no_negativo CHECK (saldo_actual >= 0);
  END IF;
END $$;

-- Migración de datos existentes: mapear el booleano `abierta` ya presente
-- a `estado`, sin inventar fecha_cierre/saldo_contado/responsable para el
-- histórico que nunca los tuvo (quedan NULL, tal como pide no inventar
-- información).
UPDATE cajas SET estado = 'ABIERTA' WHERE abierta = TRUE;
UPDATE cajas SET estado = 'CERRADA' WHERE abierta = FALSE;

-- Regla "no se puede reabrir la misma caja/canal dos veces a la vez":
-- índice único parcial sobre (descripcion) solo para las filas con
-- estado = 'ABIERTA'. Esto permite "Efectivo" y "Qr" abiertas al mismo
-- tiempo (descripciones distintas) pero impide dos filas "Efectivo"
-- ABIERTA simultáneamente. La validación se protege en el backend, pero el
-- índice es la garantía real ante peticiones concurrentes (ver punto 24 del
-- pedido: dos usuarios abriendo al mismo tiempo).
CREATE UNIQUE INDEX IF NOT EXISTS idx_cajas_unica_abierta_por_descripcion
ON cajas (descripcion)
WHERE estado = 'ABIERTA';

COMMENT ON COLUMN cajas.fecha_cierre IS 'Fecha y hora del cierre formal (arqueo). NULL mientras la caja está ABIERTA o si es un registro histórico previo a este cambio.';
COMMENT ON COLUMN cajas.saldo_esperado IS 'Saldo calculado por el sistema al momento del cierre: saldo_inicial + ingresos - egresos (movimientos_caja). NULL hasta que se cierra.';
COMMENT ON COLUMN cajas.saldo_contado IS 'Dinero físico contado por el responsable al cerrar la caja (arqueo). NULL hasta que se cierra.';
COMMENT ON COLUMN cajas.diferencia IS 'saldo_contado - saldo_esperado. Negativo = faltante, positivo = sobrante, 0 = caja cuadrada. Se conserva siempre para auditoría, nunca se oculta.';
COMMENT ON COLUMN cajas.id_admin_apertura IS 'Administrativo que abrió la caja.';
COMMENT ON COLUMN cajas.id_admin_cierre IS 'Administrativo que realizó el cierre/arqueo.';
COMMENT ON COLUMN cajas.estado IS 'ABIERTA o CERRADA. Reemplaza el uso operativo de `abierta`, que se conserva sincronizada por compatibilidad.';

COMMENT ON COLUMN asistencias.observacion IS 'Nota libre opcional, usada tipicamente para explicar una correccion o una circunstancia especial.';

COMMENT ON COLUMN asistencias.fecha_actualizacion IS 'Cuando se edito, anulo o reactivo esta asistencia por ultima vez. NULL si nunca se modifico.';

-- =============================
-- TABLA: Auditoría de Asistencias
-- =============================
-- Registro inmutable de quién hizo qué, cuándo y por qué sobre una
-- asistencia (creación, edición, anulación, reactivación). No se expone

CREATE TABLE IF NOT EXISTS auditoria_asistencias (

    id_asistencia INT NOT NULL REFERENCES asistencias(id_asistencia),
    id_admin INT REFERENCES administrativos(id_admin),
    accion VARCHAR(20) NOT NULL CHECK (accion IN ('CREACION', 'EDICION', 'ANULACION', 'REACTIVACION')),

    motivo TEXT,
    datos_anteriores JSONB,

COMMENT ON TABLE auditoria_asistencias IS 'Historial de quien creo/edito/anulo/reactivo cada asistencia, con motivo y snapshot antes/despues en JSONB. Solo lectura desde la interfaz normal.';
COMMENT ON COLUMN auditoria_asistencias.motivo IS 'Obligatorio en la app para EDICION, ANULACION y REACTIVACION. NULL para CREACION (no aplica).';
COMMENT ON COLUMN auditoria_asistencias.datos_anteriores IS 'Snapshot de la fila asistencias antes del cambio. NULL para CREACION.';
COMMENT ON COLUMN auditoria_asistencias.datos_nuevos IS 'Snapshot de la fila asistencias despues del cambio.';

-- ==========================================================
-- MODELO CAJA/JORNADA -> CANALES DE COBRO (2026-09)
-- ==========================================================
-- Antes de esto, cada canal de cobro (Efectivo, Qr, ...) se modelaba como
-- una fila INDEPENDIENTE de `cajas` (ej. "Efectivo", "Qr", "PRODUCTOS
-- EFECTIVO", "PRODUCTOS QR"), y `pagos.id_caja`/`ventas.id_caja` en
-- realidad apuntaban al canal, no a una jornada de trabajo. Esto impedia
-- separar "cuanto dinero total entro" de "cuanto efectivo fisico deberia
-- haber", y hacia que abrir/cerrar una jornada significara abrir/cerrar
-- cuatro filas sueltas en vez de una sola jornada con cuatro canales
-- dentro.
--
-- `cajas` SE MANTIENE representando una jornada de un punto de cobro (no
-- se crea una tabla nueva de "jornadas"): el cambio es que ahora el canal
-- de cobro se registra explicitamente en cada operacion/movimiento, en vez
-- de inferirse de a que fila de `cajas` pertenecia.
--
-- Aplicado por scripts/migrate-canales-cobro.js (migracion versionada,
-- registrada en la tabla `migraciones_sistema`; idempotente: si se vuelve
-- a correr y la migracion ya quedo aplicada, no repite el backfill). El
-- rollback seguro esta en scripts/rollback-canales-cobro.js: NO hace DROP
-- COLUMN a ciegas, primero verifica contra migraciones_sistema.metadata si
-- hay pagos/ventas/movimientos creados DESPUES de esta migracion que ya
-- tengan canal_cobro asignado por la app -- si los hay, se niega a
-- continuar hasta que se reconcilien, para no perder informacion nueva.

CREATE TABLE IF NOT EXISTS migraciones_sistema (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) UNIQUE NOT NULL,
    fecha_aplicada TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    metadata JSONB
);
COMMENT ON TABLE migraciones_sistema IS 'Registro de migraciones puntuales (scripts/migrate-*.js) ya aplicadas contra esta base, con metadata para permitir un rollback seguro.';

-- Catalogo unico de canales de cobro: agregar un canal nuevo en el futuro
-- es un INSERT aca, no un ALTER TABLE/CHECK repetido en pagos, ventas y
-- movimientos_caja.
CREATE TABLE IF NOT EXISTS canales_cobro (
    codigo VARCHAR(15) PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL,
    orden INT NOT NULL DEFAULT 0,
    activo BOOLEAN NOT NULL DEFAULT TRUE
);
COMMENT ON TABLE canales_cobro IS 'Catalogo unico de canales de cobro. Referenciado por pagos/ventas/movimientos_caja.canal_cobro. Ampliar un canal nuevo = INSERT aca, no ALTER TABLE en otras tablas.';

INSERT INTO canales_cobro (codigo, nombre, orden) VALUES
  ('Efectivo', 'Efectivo', 1),
  ('QR', 'QR', 2),
  ('Transferencia', 'Transferencia', 3),
  ('Tarjeta', 'Tarjeta', 4)
ON CONFLICT (codigo) DO NOTHING;

ALTER TABLE pagos ADD COLUMN IF NOT EXISTS canal_cobro VARCHAR(15) REFERENCES canales_cobro(codigo);
ALTER TABLE ventas ADD COLUMN IF NOT EXISTS canal_cobro VARCHAR(15) REFERENCES canales_cobro(codigo);
ALTER TABLE movimientos_caja ADD COLUMN IF NOT EXISTS canal_cobro VARCHAR(15) REFERENCES canales_cobro(codigo);

CREATE INDEX IF NOT EXISTS idx_pagos_canal_cobro ON pagos(canal_cobro);
CREATE INDEX IF NOT EXISTS idx_ventas_canal_cobro ON ventas(canal_cobro);
CREATE INDEX IF NOT EXISTS idx_movimientos_caja_canal_cobro ON movimientos_caja(canal_cobro);

COMMENT ON COLUMN pagos.canal_cobro IS 'Canal por el que se cobro esta fila de pago (Efectivo/QR/Transferencia/Tarjeta). En un pago mixto, cada fila tiene su propio canal.';
COMMENT ON COLUMN ventas.canal_cobro IS 'Canal por el que se cobro esta venta.';
COMMENT ON COLUMN movimientos_caja.canal_cobro IS 'Canal afectado por ESTE movimiento puntual (Ingreso o Egreso). Se deriva del id_caja que tenia el movimiento en el momento en que se creo, no del id_caja actual del pago/venta relacionado (que puede haber cambiado despues por una edicion). Un Egreso en Efectivo reduce el efectivo fisico del arqueo; un Egreso en QR/Transferencia/Tarjeta no.';

-- No se agrega columna de "tipo_operacion": se deriva de
-- movimientos_caja.origen (Pago -> Membresia, Venta -> Producto,
-- Desembolso/Reembolso/Otro -> el movimiento correspondiente), que ya
-- existia y ya distingue esto sin necesidad de duplicar el dato.

-- Backfill de datos historicos: mapeo verificado contra un diagnostico de
-- solo lectura (scripts/diagnostico-canales-caja.js) antes de tocar nada.
-- Las 4 cajas existentes en ese momento eran inequivocas por descripcion:
--   id_caja 1 "Efectivo"           -> Efectivo
--   id_caja 2 "Qr"                 -> QR
--   id_caja 3 "PRODUCTOS EFECTIVO" -> Efectivo
--   id_caja 4 "PRODUCTOS QR"       -> QR
-- Cada tabla se backfillea con SU PROPIO id_caja (no cruzado contra el
-- id_caja actual del pago/venta relacionado): un movimiento historico
-- puede legitimamente tener un id_caja distinto al id_caja actual de su
-- pago/venta cuando ese pago fue editado despues (ver
-- src/app/api/pagos/[id]/route.js, PUT: mover un pago de caja crea un
-- Egreso "Reembolso" en la caja vieja + un Ingreso "Pago" en la caja
-- nueva, ambos con el mismo id_referencia). Confirmado con
-- scripts/_check-consistencia-caja.js antes de migrar: 23 movimientos de
-- origen Pago y 1 de origen Venta muestran esa diferencia esperada, no
-- corrupcion.

-- cajas.fecha_apertura pasa de DATE a TIMESTAMP para que una jornada
-- profesional pueda registrar la hora real de apertura, no solo el dia.
ALTER TABLE cajas ALTER COLUMN fecha_apertura TYPE TIMESTAMP USING fecha_apertura::timestamp;
COMMENT ON COLUMN cajas.fecha_apertura IS 'Fecha y hora de apertura de la jornada. Para las jornadas creadas antes de esta migracion, la columna era DATE (sin hora): el valor 00:00:00 en esas filas NO es una hora real de apertura, es un artefacto de la conversion de tipo -- solo la fecha es confiable para ese historico (id_caja 1-4). Ver migraciones_sistema.metadata (canal_cobro_2026_09) para la lista exacta.';
